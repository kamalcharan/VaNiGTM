# Run on the VPS over SSH (bash), not in PowerShell.

# ---- 1. STANDARD DEPLOY ------------------------------------------------------
cd /opt/vikuna/src/vanigtm
git status --short
git rev-parse HEAD > ~/vani-previous-sha
git fetch origin && git checkout main && git pull origin main
git log --oneline -1
bash deploy/vani-main-vps/deploy-vani.sh

# ---- 2. MIGRATIONS -- right after step 1, before anyone uses the console -------
# (new code + old schema = errors like: column "..." does not exist)
docker exec vani-backend node dist/migrate.js --status
docker exec vani-backend node dist/migrate.js
docker exec vani-backend node dist/migrate.js --status | tail -3

# ---- 3. NGINX (only when deploy/vani-main-vps/*.conf or *.inc changed) -------
F=/opt/vikuna/docker/docker/config/nginx/conf.d
cp "$F/api.vikuna.io.conf" "$F/api.vikuna.io.conf.bak-$(date +%Y%m%d-%H%M)"
cp deploy/vani-main-vps/api.vikuna.io.conf deploy/vani-main-vps/vani-cors.inc "$F/"
docker exec vikuna-nginx nginx -t && docker exec vikuna-nginx nginx -s reload

# ---- 4. VERIFY ---------------------------------------------------------------
docker ps --filter name=vani --format '{{.Names}}\t{{.Status}}'
docker logs --tail 30 vani-backend
docker logs --tail 30 vani-worker
docker logs vani-worker 2>&1 | grep -c "Poll error"
curl -s https://api.vikuna.io/health
curl -s -X POST https://api.vikuna.io/api/v1/embed/boot -H 'Content-Type: application/json' -d '{}'

# ---- 5. ROLLBACK -------------------------------------------------------------
cd /opt/vikuna/src/vanigtm
git checkout "$(cat ~/vani-previous-sha)"
bash deploy/vani-main-vps/deploy-vani.sh
git checkout main


# =============================================================================
# RELEASE 2026-10-02b — uploads are temporary (ETL_UPLOAD_DIR, shared folder)
# No migrations. Order: A, then 1, B, 4.
# =============================================================================

# ---- A. BEFORE step 1 --------------------------------------------------------
# A1. The shared temp folder on the host.
mkdir -p /opt/vikuna/data/vani-uploads

# A2. Two settings (skips any already set).
COMPOSE_DIR=$(docker inspect vani-backend --format '{{index .Config.Labels "com.docker.compose.project.working_dir"}}')
cp "$COMPOSE_DIR/.env" "$COMPOSE_DIR/.env.bak-$(date +%Y%m%d-%H%M)"
grep -q '^ETL_UPLOAD_DIR='            "$COMPOSE_DIR/.env" || echo 'ETL_UPLOAD_DIR=/app/uploads'       >> "$COMPOSE_DIR/.env"
grep -q '^ETL_UPLOAD_TEMP_TTL_HOURS=' "$COMPOSE_DIR/.env" || echo 'ETL_UPLOAD_TEMP_TTL_HOURS=24'      >> "$COMPOSE_DIR/.env"
grep -E '^ETL_UPLOAD_(DIR|TEMP_TTL_HOURS)=' "$COMPOSE_DIR/.env"

# A3. Mount the folder in BOTH services of the live compose file.
#     This prints the file(s) compose uses:
docker inspect vani-backend --format '{{index .Config.Labels "com.docker.compose.project.config_files"}}'
#     Open it (nano <that path>) and add, under vani-backend: AND under vani-worker:
#       volumes:
#         - /opt/vikuna/data/vani-uploads:/app/uploads
#     (if a service already has volumes:, add only the "- ..." line to it).
#     The repo's deploy/vani-main-vps/docker-compose.vani.yml shows the result.

# ---- B. AFTER step 1: both containers see the same folder --------------------
docker inspect vani-backend --format '{{range .Mounts}}{{.Source}} -> {{.Destination}}{{println}}{{end}}'
docker inspect vani-worker  --format '{{range .Mounts}}{{.Source}} -> {{.Destination}}{{println}}{{end}}'
docker logs vani-backend 2>&1 | grep "ETL:" | tail -1
docker logs vani-worker  2>&1 | grep "ETL:" | tail -1
# Then in the console: import a small CSV, stage it; afterwards this is empty:
ls -la /opt/vikuna/data/vani-uploads
